## SSE 事件格式
每帧：`{ seq, time, runId, type, ... }`，`type` 取值：
- `snapshot`：连接建立时先推一帧现状，含 `{ status, summary, results }`
- `run:start` `{ run: {id,name,total,projectId} }`
- `run:log` `{ level: 'info'|'warn'|'error', message }`
- `case:start` `{ caseId, name, type, index, total }`
- `step:start` `{ caseId, stepId, name }`
- `step:log` `{ caseId, stepId, level, message }`
- `assertion` `{ caseId, stepId, result: AssertionResult }`
- `step:end` `{ caseId, stepId, status, durationMs }`
- `case:end` `{ caseId, name, status, durationMs, summary, result }`
- `run:end` `{ status, summary, durationMs }`

  op: 'equals'|'notEquals'|'contains'|'notContains'|'matches'|'gt'|'gte'|'lt'|'lte'|'exists'|'truthy'

  id: string; name: string; screenshot?: boolean; continueOnFail?: boolean
  request?: { method: string; url: string; headers?: Record<string,string>; query?: Record<string,string>; bodyMode?: 'none'|'json'|'text'|'form'; body?: string; timeoutMs?: number }
  extract?: { name: string; from: 'json'|'header'|'text'; path?: string; regex?: string; group?: number }[]
  assertions?: Assertion[]
}

export type WebStep = {
  id: string; name: string; screenshot?: boolean; continueOnFail?: boolean
  action?: { type: 'goto'|'fill'|'click'|'select'|'press'|'waitForSelector'|'waitForTimeout'|'screenshot'|'hover'|'check'; selector?: string; value?: string; url?: string; ms?: number }
  assertions?: Assertion[]
}

export type TestCase = {
  id: string; projectId: string; name: string; type: 'api'|'web'; description: string
  tags: string[]; enabled: boolean; variables: Record<string,string>; timeoutMs?: number
  steps: (ApiStep|WebStep)[]; createdAt: number; updatedAt: number
};

export type AssertionResult = { assertionId: string; type: string; op: string; label: string; expected: string; actual: string; ok: boolean; message: string }

export type StepResult = {
  stepId: string; name: string; status: 'passed'|'failed'|'error'|'skipped'; durationMs: number
  request?: { method: string; url: string; headers?: Record<string,string>; body?: string }
  response?: { status: number; headers: Record<string,string>; bodyPreview: string; bodySize: number; truncated?: boolean }
  assertions: AssertionResult[]; logs: string[]; screenshots: string[]; error?: string; extracted?: Record<string,string>; pageUrl?: string
}

export type CaseResult = { caseId: string; name: string; type: 'api'|'web'; tags: string[]; status: string; durationMs: number; steps: StepResult[]; logs: string[]; screenshots: string[]; error?: string }

export type RunSummary = {
  id: string; projectId: string; projectName?: string; name: string; status: string
  startedAt: number; finishedAt: number|null; durationMs: number|null
  summary: { total: number; passed: number; failed: number; error: number; skipped: number; durationMs: number }
  cases: { caseId: string; name: string; type: string; status: string; durationMs: number; failedAssertions: number; screenshots: string[] }[]
}

export type Run = RunSummary & { results: CaseResult[]; active?: boolean; variables?: Record<string,string>; error?: string }
